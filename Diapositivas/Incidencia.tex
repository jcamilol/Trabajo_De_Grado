\begin{frame}{Propiedades: compacidad}
   \begin{prop}
      La topología de incidencia de un grafo es \alert{compacta} si y solo si el grafo es \alert{finito}.\vspace{0.3cm}
   \end{prop}
\end{frame}
\begin{frame}{Propiedades: isomorfismo de grafos}
   \begin{prop}
      Si dos grafos son \alert{isomorfos}, sus respectivos espacios topológicos de incidencia son \alert{homeomorfos}.\vspace{0.3cm}
   \end{prop}
\end{frame}

\begin{frame}{Propiedades: conexidad}
   \begin{prop}
      Si un grafo es \alert{disconexo}, su topología de incidencia es \alert{disconexa}.\vspace{0.3cm}
   \end{prop}
\end{frame}
\begin{frame}{Propiedades: conexidad}
   \begin{prop}
      Sea $G=(V,E)$ un grafo
      \begin{itemize}
         \item conexo,
         \item con topología de incidencia no discreta,
         \item con $4$ o más vértices.
      \end{itemize}
      \uncover<2->{Entonces su topología de incidencia es \alert{conexa si y solo si} \alert{$e\cap f\neq\emptyset$} para cualesquiera $e,f\in E$ con $e\neq f$.}\vspace{0.3cm}
   \end{prop}
\end{frame}
