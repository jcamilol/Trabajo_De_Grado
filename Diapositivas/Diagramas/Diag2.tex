\begin{tikzpicture}[
  abierto/.style={draw=black, thick, dashed}
]
  % Abiertos que contienen a x
  \def\abiertoA{[rotate around={20:(-0.8,0.35)}] (-0.8,0.35) ellipse (1.7 and 1.0)}
