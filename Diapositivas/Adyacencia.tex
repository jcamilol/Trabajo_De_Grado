\begin{frame}{Propiedades: topologías discretas}
   \begin{prop}

\begin{frame}{Propiedades: compacidad}
   \begin{prop}
      La topología de adyacencia de un grafo es \alert{compacta} si y solo si el grafo es \alert{finito}.\vspace{0.3cm}
   \end{prop}
\end{frame}

\begin{frame}{Isomorfismo de grafos}
   \begin{itemize}
      \item $G=(V,E)$ y $G'=(V',E')$ dos grafos
      \item $\varphi: V\to V'$ una biyección
      \item $\varphi$ es un \alert{isomorfismo} si y solo si vértices adyacentes de $G$ son adyacentes en $G'$
   \end{itemize}
