\begin{frame}{Propiedades: compacidad}
   \begin{prop}
      La topología de adyacencia de un grafo es \alert{compacta} si y solo si el grafo es \alert{finito}.\vspace{0.3cm}
   \end{prop}
\end{frame}

\begin{frame}{Isomorfismo de grafos}
   \begin{itemize}
      \item $G=(V,E)$ y $G'=(V',E')$ dos grafos
      \item $\varphi: V\to V'$ una biyección
      \item $\varphi$ es un \alert{isomorfismo} si y solo si vértices adyacentes de $G$ son adyacentes en $G'$
   \end{itemize}

\begin{frame}{Densidad}
   \begin{Def}
      Un subconjunto $A\subseteq X$ se dice \alert{denso} si $\overline{A}=X$.\vspace{0.3cm}
   \end{Def}
   \vspace{0.3cm}
   \uncover<2->{%
   \begin{Def}
      Si $A$ es denso en $X$, decimos que $A$ es \alert{denso minimal} si ningún otro subconjunto denso está propiamente contenido en $A$.\vspace{0.3cm}
   \end{Def}}
\end{frame}

\begin{frame}{Propiedades: densidad y número cromático}
   \begin{prop}
      Si $A$ es un subconjunto \alert{denso minimal} de $V$, entonces $\chi(G)\leq |A|$.\vspace{0.3cm}
   \end{prop}
